using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Runtime.CompilerServices;
using UnityEngine;
using Object = UnityEngine.Object;

namespace QBS.Core
{
	public enum LogLevel
	{
		Trace,
		Debug,
		Info,
		Warning,
		Error,
		Fatal,
	}

	public static partial class Log
	{
		private const string AssertionFailedTag = "ASSERTION FAILED";

		/// <summary>
		///     The console sink every game starts with, exposed so a game that wants its output somewhere
		///     else can <see cref="RemoveSink" /> it. Logging therefore needs no setup, and needs no
		///     guesswork to undo.
		/// </summary>
		public static readonly UnitySink DefaultSink = new UnitySink();

		private static readonly List<ILogSink> _sinks = new() { DefaultSink };

		public static LogLevel MinimumLevel { get; set; } = LogLevel.Trace;

		/// <summary>
		///     Adds a destination for every event that passes <see cref="MinimumLevel" /> and whose channel is
		///     enabled. Adding the same instance twice does nothing.
		/// </summary>
		public static void AddSink(ILogSink sink)
		{
			if (sink == null || _sinks.Contains(sink))
			{
				return;
			}

			_sinks.Add(sink);
		}

		public static void RemoveSink(ILogSink sink)
		{
			_sinks.Remove(sink);
		}

		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void SetChannelEnabled(LogCategory channel, bool enabled)
		{
			if (channel == null)
			{
				return;
			}

			//A group writes rules for its members, as clearing a combination cleared its bits.
			LogRules.SetCategory(channel, enabled);
		}

		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static bool IsChannelEnabled(LogCategory channel)
		{
			return channel != null && channel.Enabled;
		}

		/// <summary>Drops every channel rule and switches all channels on or off.</summary>
		public static void SetAllChannelsEnabled(bool enabled)
		{
			LogRules.Reset(enabled);
		}

		[Conditional("ENABLE_LOGS")]
		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Trace(string message, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Trace, message, tag, color, channel, context, null, member, file, line));
		}

		[Conditional("ENABLE_LOGS")]
		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Debug(string message, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Debug, message, tag, color, channel, context, null, member, file, line));
		}

		[Conditional("ENABLE_LOGS")]
		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Info(string message, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Info, message, tag, color, channel, context, null, member, file, line));
		}

		// String allocation
		[Conditional("ENABLE_LOGS")]
		[HideInCallstack]
		public static void InfoSnipe(string message, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Info, $"( -_•)▄︻デ══━一 {message}", tag, color, channel, context, null, member, file, line));
		}

		[Conditional("ENABLE_LOGS")]
		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Warning(string message, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Warning, message, tag, color, channel, context, null, member, file, line));
		}

		//Error and Fatal alone are not [Conditional]: the attribute strips the call at the call site, so a
		//player built without ENABLE_LOGS would report nothing to a crash reporter at all.
		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Error(string message, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Error, message, tag, color, channel, context, null, member, file, line));
		}

		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Error(string message, System.Exception exception, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Error, message, tag, color, channel, context, exception, member, file, line));
		}

		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Fatal(string message, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Fatal, message, tag, color, channel, context, null, member, file, line));
		}

		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Fatal(string message, System.Exception exception, string tag = null, Color color = default,
			LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Fatal, message, tag, color, channel, context, exception, member, file, line));
		}

		/// <summary>
		///     Reports a caught exception at <see cref="LogLevel.Error" />. The exception object rides along
		///     whole, so a crash reporter can capture it with its own stack rather than a copy of the text.
		/// </summary>
		[HideInCallstack]
		[MethodImpl(MethodImplOptions.AggressiveInlining)]
		public static void Exception(System.Exception exception, string message = null, string tag = null,
			Color color = default, LogCategory channel = null, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			Dispatch(new LogEvent(LogLevel.Error, message ?? exception?.Message, tag, color, channel, context, exception, member, file, line));
		}

		[HideInCallstack]
		[Conditional("ENABLE_LOGS")]
		private static void Dispatch(in LogEvent logEvent)
		{
			if (logEvent.Level < MinimumLevel)
			{
				return;
			}

			if (logEvent.Channel != null && !logEvent.Channel.Enabled)
			{
				return;
			}

			//Indexed rather than foreach: a sink is free to add or remove one while handling an event,
			//and the enumerator would throw where this simply picks the change up on the next call.
			for (var i = 0; i < _sinks.Count; i++)
			{
				_sinks[i].Write(in logEvent);
			}
		}

		// Assert methods
		[Conditional("ENABLE_LOGS")]
		[HideInCallstack]
		public static void Assert(bool condition, string message, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			if (!condition)
			{
				Dispatch(new LogEvent(LogLevel.Error, message, AssertionFailedTag, Color.red, null, context, null, member, file, line));
			}
		}

		[HideInCallstack]
		public static void AssertThrow(bool condition, string message, Object context = null,
			[CallerMemberName] string member = "", [CallerFilePath] string file = "", [CallerLineNumber] int line = 0)
		{
			if (!condition)
			{
				Dispatch(new LogEvent(LogLevel.Fatal, $"[ASSERTION FAILED] {message}", AssertionFailedTag, Color.red, null, context, null, member, file, line));
				throw new System.Exception($"Assertion failed: {message}");
			}
		}

		internal static string ToStringFast(this LogLevel level)
		{
			return level switch
			{
				LogLevel.Trace => nameof(LogLevel.Trace),
				LogLevel.Debug => nameof(LogLevel.Debug),
				LogLevel.Info => nameof(LogLevel.Info),
				LogLevel.Warning => nameof(LogLevel.Warning),
				LogLevel.Error => nameof(LogLevel.Error),
				LogLevel.Fatal => nameof(LogLevel.Fatal),
				_ => "",
			};
		}
	}


}
