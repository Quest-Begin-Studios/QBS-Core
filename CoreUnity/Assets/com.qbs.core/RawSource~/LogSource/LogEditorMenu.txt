using System.Collections.Generic;
using UnityEditor;
using UnityEngine;

namespace QBS.Core.Editor
{
	public static class LogEditorMenu
	{
		private const string MenuRoot = "Tools/Logs/";
		private const string ConfigureMenu = MenuRoot + "Configure Logging";

		[MenuItem(ConfigureMenu, false, 1)]
		private static void OpenLogConfiguration()
		{
			LogConfigWindow.ShowWindow();
		}

		[InitializeOnLoadMethod]
		private static void InitializeLogSettings()
		{
			RestoreMinimumLogLevel();
			RestoreRules();
			WarnUserAboutInactiveLogs();

			EditorApplication.quitting -= OnQuit;
			EditorApplication.quitting += OnQuit;

			EditorApplication.playModeStateChanged -= OnStateChange;
			EditorApplication.playModeStateChanged += OnStateChange;
		}

		private static void OnStateChange(PlayModeStateChange state)
		{
			if (state != PlayModeStateChange.ExitingPlayMode)
			{
				return;
			}

			LogEditorUtility.WriteRules(LogRules.All);
			EditorPrefs.SetInt(LogEditorConstants.MinimumLevel, (int)Log.MinimumLevel);
		}

		private static void WarnUserAboutInactiveLogs()
		{
			var areLogsEnabled = LogEditorUtility.AreLogsEnabled();
			var warningShown = EditorPrefs.GetBool(LogEditorConstants.LogsDisabledWarningShown, false);
			if (areLogsEnabled || warningShown)
			{
				return;
			}

			Debug.LogError
			(
				"Logging is disabled. Enable it in the "
				+ "Log Configuration window to utilize the logging utility."
				+ " \nTools -> Logs -> Configure Logging "
			);

			EditorPrefs.SetBool(LogEditorConstants.LogsDisabledWarningShown, true);

		}

		private static void RestoreMinimumLogLevel()
		{
			if (EditorPrefs.HasKey(LogEditorConstants.MinimumLevel))
			{
				var level = (LogLevel)EditorPrefs.GetInt(LogEditorConstants.MinimumLevel);
				Log.MinimumLevel = level;
			}
			else
			{
				Log.MinimumLevel = LogLevel.Trace;
			}
		}

		private static void RestoreRules()
		{
			//Nothing saved on this machine yet: start from the project's shipped settings, so a fresh checkout
			//sees the channels the build will rather than everything on.
			var rules = (IEnumerable<LogRule>)LogEditorUtility.ReadRules() ?? LogEditorUtility.LoadRuntimeSettings()?.Rules;
			LogRules.Load(rules);
		}

		private static void OnQuit()
		{
			EditorPrefs.SetBool(LogEditorConstants.LogsDisabledWarningShown, false);
		}
	}
}
