using UnityEngine;

namespace QBS.Core
{
	[CreateAssetMenu(fileName = "RuntimeLogSettings", menuName = "QBS/Runtime Log Settings")]
	public class RuntimeLogSettings : ScriptableObject
	{
		[SerializeField]
		[Tooltip("Minimum log level to display. Logs below this level will be filtered out.")]
		private LogLevel minimumLevel = LogLevel.Trace;

		//Stored as the long it is: Unity serializes an enum field as 32 bits, which would drop every
		//channel past the 32nd now that LogChannel is long-backed.
		[SerializeField]
		[Tooltip("Enabled log channels, as a bit mask. Edit from Tools > QBS > Logs > Configure Logging.")]
		private long enabledChannels = (long) LogChannel.All;

		public LogLevel MinimumLevel
		{
			get => minimumLevel;
			set => minimumLevel = value;
		}

		public LogChannel EnabledChannels
		{
			get => (LogChannel) enabledChannels;
			set => enabledChannels = (long) value;
		}
	}
}
