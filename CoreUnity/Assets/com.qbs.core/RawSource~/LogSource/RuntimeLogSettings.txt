using System.Collections.Generic;
using UnityEngine;

namespace QBS.Core
{
	[CreateAssetMenu(fileName = "RuntimeLogSettings", menuName = "QBS/Runtime Log Settings")]
	public class RuntimeLogSettings : ScriptableObject
	{
		[SerializeField]
		[Tooltip("Minimum log level to display. Logs below this level will be filtered out.")]
		private LogLevel minimumLevel = LogLevel.Trace;

		//Rules by name: nothing here depends on which categories exist or the order they were created in, and a
		//rule for a package that has since been removed simply matches nothing.
		[SerializeField]
		[Tooltip("Which log channels are on, as prefix rules. Edit from Tools > QBS > Logs > Configure Logging.")]
		private List<LogRule> rules = new() { new LogRule(LogRules.Root, true) };

		public LogLevel MinimumLevel
		{
			get => minimumLevel;
			set => minimumLevel = value;
		}

		public IReadOnlyList<LogRule> Rules
		{
			get => rules;
			set => rules = new List<LogRule>(value);
		}
	}
}
