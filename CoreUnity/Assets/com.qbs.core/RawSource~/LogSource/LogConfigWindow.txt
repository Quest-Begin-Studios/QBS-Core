using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEngine;

namespace QBS.Core.Editor
{
	public class LogConfigWindow : EditorWindow
	{
		private const string BuiltInSection = "Built-in";
		private const string GroupsSection = "Groups";
		private const float PillHeight = 22f;
		private const float PillSpacing = 4f;
		private const float PillPadding = 20f;
		private const float LevelLabelWidth = 96f;

		//Card padding and margins plus room for the vertical scrollbar.
		private const float ContentInset = 48f;

		private static readonly string[] LevelNames = Enum.GetNames(typeof(LogLevel));

		private bool _logsEnabled;
		private LogLevel _minimumLevel;
		private RuntimeLogSettings _runtimeSettings;

		private Vector2 _channelScrollPosition;
		private Vector2 _windowScrollPosition;
		private List<LogCategory> _individualChannels;
		private List<LogCategory> _compositeChannels;
		private List<(string Name, List<LogCategory> Channels)> _sections;
		private HashSet<LogCategory> _editorOnlyChannels;
		private List<string> _problems;
		private string _search = "";
		private readonly HashSet<string> _collapsedSections = new();

		private Texture2D _cardTexture;
		private GUIStyle _titleStyle;
		private GUIStyle _cardStyle;
		private GUIStyle _sectionFoldoutStyle;
		private GUIStyle _countStyle;
		private GUIStyle _subtleStyle;
		private GUIStyle _pillLabelStyle;

		public static void ShowWindow()
		{
			var window = GetWindow<LogConfigWindow>("Log Configuration");
			window.minSize = new Vector2(420, 480);
			window.Show();
		}

		private void OnEnable()
		{
			wantsMouseMove = true;

			// Load or create RuntimeLogSettings
			_runtimeSettings = LogEditorUtility.LoadOrCreateRuntimeSettings();

			// Nothing saved yet leaves the rules as they are, which is everything on.
			var savedRules = LogEditorUtility.ReadRules();
			if (savedRules != null)
			{
				LogRules.Load(savedRules);
			}

			_minimumLevel = (LogLevel)EditorPrefs.GetInt(LogEditorConstants.MinimumLevel, 0);
			_logsEnabled = LogEditorUtility.AreLogsEnabled();

			// Sync RuntimeLogSettings with current EditorPrefs
			SyncRuntimeSettingsFromEditorPrefs();

			RefreshChannels();
		}

		private void OnDisable()
		{
			if (_cardTexture != null)
			{
				DestroyImmediate(_cardTexture);
			}
		}

		private void InitializeStyles()
		{
			//The texture goes with the window on a domain reload while the styles can survive, so check both.
			if (_cardStyle != null && _cardTexture != null)
			{
				return;
			}

			_cardTexture = new Texture2D(1, 1) { hideFlags = HideFlags.HideAndDontSave };
			_cardTexture.SetPixel(0, 0, Palette.Card);
			_cardTexture.Apply();

			_titleStyle = new GUIStyle(EditorStyles.boldLabel)
			{
				fontSize = 16,
				fixedHeight = PillHeight,
				alignment = TextAnchor.MiddleLeft,
			};

			_cardStyle = new GUIStyle
			{
				normal = { background = _cardTexture },
				padding = new RectOffset(10, 10, 8, 10),
				margin = new RectOffset(6, 6, 4, 4),
			};

			_sectionFoldoutStyle = new GUIStyle(EditorStyles.foldout)
			{
				fontStyle = FontStyle.Bold,
			};

			_countStyle = new GUIStyle(EditorStyles.miniLabel)
			{
				alignment = TextAnchor.MiddleRight,
			};

			_subtleStyle = new GUIStyle(EditorStyles.miniLabel)
			{
				wordWrap = true,
			};

			_pillLabelStyle = new GUIStyle(EditorStyles.label)
			{
				alignment = TextAnchor.MiddleCenter,
				fontSize = 11,
				clipping = TextClipping.Clip,
			};
		}

		private void OnGUI()
		{
			InitializeStyles();

			if (Event.current.type == EventType.MouseMove)
			{
				Repaint();
			}

			_windowScrollPosition = EditorGUILayout.BeginScrollView(_windowScrollPosition);

			GUILayout.Space(4);
			DrawHeader();
			DrawToolbar();
			DrawChannelsSection();
			DrawCompositeChannels();
			DrawValidationSection();
			GUILayout.Space(6);

			EditorGUILayout.EndScrollView();
		}

		private void DrawHeader()
		{
			EditorGUILayout.BeginVertical(_cardStyle);

			EditorGUILayout.BeginHorizontal();
			GUILayout.Label("QBS Logging", _titleStyle);
			GUILayout.FlexibleSpace();
			DrawEnableLogsToggle();
			EditorGUILayout.EndHorizontal();

			GUILayout.Space(8);
			DrawMinimumLevelSection();

			GUILayout.Space(6);
			var activeCount = _individualChannels.Count(Log.IsChannelEnabled);
			var summary = $"{activeCount} of {_individualChannels.Count} channels on";
			if (_problems.Count > 0)
			{
				summary += $"  ·  {_problems.Count} issue{(_problems.Count == 1 ? "" : "s")} below";
			}
			GUILayout.Label(summary, _subtleStyle);

			if (!_logsEnabled)
			{
				GUILayout.Space(4);
				EditorGUILayout.HelpBox
				(
					"ENABLE_LOGS is off for this build target, so Trace to Warning calls are compiled out. Error and Fatal still log.",
					MessageType.Warning
				);
			}

			EditorGUILayout.EndVertical();
		}

		private void DrawEnableLogsToggle()
		{
			var isCompiling = EditorApplication.isCompiling;
			var text = isCompiling ? "Compiling…" : _logsEnabled ? "● Logs on" : "○ Logs off";
			var content = new GUIContent(text, "Adds or removes the ENABLE_LOGS scripting define symbol for the selected build target.");

			EditorGUI.BeginDisabledGroup(isCompiling);
			if (Pill(content, _logsEnabled ? Palette.On : Palette.Danger, false, 104f))
			{
				_logsEnabled = !_logsEnabled;
				ToggleEnableLogs();
			}
			EditorGUI.EndDisabledGroup();
		}

		private void DrawMinimumLevelSection()
		{
			EditorGUILayout.BeginHorizontal();
			GUILayout.Label(new GUIContent("Minimum level", "Only logs at or above this level are shown."), GUILayout.Width(LevelLabelWidth), GUILayout.Height(PillHeight));

			var available = position.width - ContentInset - LevelLabelWidth;
			var width = Mathf.Max(40f, (available - PillSpacing * (LevelNames.Length - 1)) / LevelNames.Length);

			for (var i = 0; i < LevelNames.Length; i++)
			{
				var level = (LogLevel)i;
				var levelColor = GetLevelColor(level);
				var fill = level == _minimumLevel ? levelColor
					: level > _minimumLevel ? Color.Lerp(Palette.Off, levelColor, 0.35f)
					: Palette.Off;

				var content = new GUIContent(LevelNames[i], $"Show {LevelNames[i]} and above.");
				if (Pill(content, fill, false, width) && level != _minimumLevel)
				{
					_minimumLevel = level;
					WriteMinimumLevel(level);
				}

				if (i < LevelNames.Length - 1)
				{
					GUILayout.Space(PillSpacing);
				}
			}

			EditorGUILayout.EndHorizontal();
		}

		private void DrawToolbar()
		{
			EditorGUILayout.BeginHorizontal(EditorStyles.toolbar);
			GUILayout.Space(4);
			_search = GUILayout.TextField(_search, EditorStyles.toolbarSearchField, GUILayout.ExpandWidth(true));

			if (GUILayout.Button(new GUIContent("All on", "Switches every channel on."), EditorStyles.toolbarButton, GUILayout.Width(56)))
			{
				SetStateOfAllChannels(true);
			}
			if (GUILayout.Button(new GUIContent("All off", "Switches every channel off."), EditorStyles.toolbarButton, GUILayout.Width(56)))
			{
				SetStateOfAllChannels(false);
			}
			EditorGUILayout.EndHorizontal();
			GUILayout.Space(2);
		}

		private void DrawChannelsSection()
		{
			foreach (var (sectionName, channels) in _sections)
			{
				var visibleChannels = channels.Where(MatchesSearch).ToList();
				if (visibleChannels.Count == 0)
				{
					continue;
				}

				EditorGUILayout.BeginVertical(_cardStyle);

				var activeCount = channels.Count(Log.IsChannelEnabled);
				if (DrawSectionHeader(sectionName, $"{activeCount} / {channels.Count} on"))
				{
					GUILayout.Space(4);
					DrawIndividualChannels(sectionName, visibleChannels);
				}

				EditorGUILayout.EndVertical();
			}
		}

		private void DrawIndividualChannels(string sectionName, List<LogCategory> channels)
		{
			var pills = channels.Select(channel => new PillItem
			(
				channel,
				new GUIContent(GetShortName(sectionName, channel), GetRuleTooltip(channel)),
				Log.IsChannelEnabled(channel) ? Palette.On : Palette.Off,
				_editorOnlyChannels.Contains(channel)
			));

			DrawPillFlow(pills, channel =>
			{
				Log.SetChannelEnabled(channel, !Log.IsChannelEnabled(channel));
				WriteRules();
			});
		}

		private void DrawCompositeChannels()
		{
			var visibleGroups = _compositeChannels.Where(MatchesSearch).ToList();
			if (visibleGroups.Count == 0)
			{
				return;
			}

			EditorGUILayout.BeginVertical(_cardStyle);

			if (DrawSectionHeader(GroupsSection, $"{_compositeChannels.Count} groups"))
			{
				GUILayout.Space(4);

				var pills = visibleGroups.Select(group =>
				{
					var activeCount = group.Members.Count(Log.IsChannelEnabled);
					var fill = activeCount == group.Members.Count ? Palette.On
						: activeCount > 0 ? Palette.Partial
						: Palette.Off;

					var tooltip = $"{group.Name}: {string.Join(", ", group.Members.Select(m => m.Name))}";
					var content = new GUIContent($"{group.Name}  {activeCount}/{group.Members.Count}", tooltip);
					return new PillItem(group, content, fill, _editorOnlyChannels.Contains(group));
				});

				//A group that is fully on switches off; any other state switches it fully on.
				DrawPillFlow(pills, group =>
				{
					var isFullyEnabled = group.Members.All(Log.IsChannelEnabled);
					Log.SetChannelEnabled(group, !isFullyEnabled);
					WriteRules();
				});

				GUILayout.Space(4);
				GUILayout.Label("Green: every member on. Amber: some on. Grey: none on.", _subtleStyle);
			}

			EditorGUILayout.EndVertical();
		}

		//Checked whenever the window is enabled, which includes after every recompile, so nothing needs a button.
		private void DrawValidationSection()
		{
			if (_problems.Count == 0)
			{
				return;
			}

			EditorGUILayout.BeginVertical(_cardStyle);

			GUILayout.Label($"⚠ {_problems.Count} issue{(_problems.Count == 1 ? "" : "s")} with declared channels", EditorStyles.boldLabel);

			foreach (var problem in _problems)
			{
				EditorGUILayout.HelpBox(problem, MessageType.Warning);
			}

			EditorGUILayout.EndVertical();
		}

		/// <summary>A foldout with a count on the right. Returns whether the section is open.</summary>
		private bool DrawSectionHeader(string sectionName, string count)
		{
			var rect = GUILayoutUtility.GetRect(0f, EditorGUIUtility.singleLineHeight, GUILayout.ExpandWidth(true));
			const float countWidth = 90f;

			var isOpen = !_collapsedSections.Contains(sectionName);
			var foldoutRect = new Rect(rect.x, rect.y, rect.width - countWidth, rect.height);
			var nowOpen = EditorGUI.Foldout(foldoutRect, isOpen, sectionName, true, _sectionFoldoutStyle);
			GUI.Label(new Rect(rect.xMax - countWidth, rect.y, countWidth, rect.height), count, _countStyle);

			if (nowOpen != isOpen)
			{
				if (nowOpen)
				{
					_collapsedSections.Remove(sectionName);
				}
				else
				{
					_collapsedSections.Add(sectionName);
				}
			}

			//While searching, every section with a match stays open so the match is visible.
			return nowOpen || _search.Length > 0;
		}

		private readonly struct PillItem
		{
			public readonly LogCategory Channel;
			public readonly GUIContent Content;
			public readonly Color Fill;
			public readonly bool Italic;

			public PillItem(LogCategory channel, GUIContent content, Color fill, bool italic)
			{
				Channel = channel;
				Content = content;
				Fill = fill;
				Italic = italic;
			}
		}

		//Lays pills out left to right, wrapping to a new row when the next one would overflow the card.
		private void DrawPillFlow(IEnumerable<PillItem> pills, Action<LogCategory> onClick)
		{
			var available = position.width - ContentInset;
			var rowWidth = 0f;
			LogCategory clicked = null;

			EditorGUILayout.BeginHorizontal();

			foreach (var pill in pills)
			{
				_pillLabelStyle.fontStyle = pill.Italic ? FontStyle.Italic : FontStyle.Normal;
				var width = Mathf.Max(56f, _pillLabelStyle.CalcSize(pill.Content).x + PillPadding);

				if (rowWidth > 0f && rowWidth + width > available)
				{
					GUILayout.FlexibleSpace();
					EditorGUILayout.EndHorizontal();
					GUILayout.Space(PillSpacing);
					EditorGUILayout.BeginHorizontal();
					rowWidth = 0f;
				}

				if (Pill(pill.Content, pill.Fill, pill.Italic, width))
				{
					clicked = pill.Channel;
				}

				GUILayout.Space(PillSpacing);
				rowWidth += width + PillSpacing;
			}

			GUILayout.FlexibleSpace();
			EditorGUILayout.EndHorizontal();

			//Applied after the loop, so the frame that handled the click drew every pill from the same state.
			if (clicked != null)
			{
				onClick(clicked);
			}
		}

		/// <summary>A flat, coloured button whose text contrasts with its fill. Returns true when clicked.</summary>
		private bool Pill(GUIContent content, Color fill, bool italic, float width)
		{
			var rect = GUILayoutUtility.GetRect(width, PillHeight, GUILayout.Width(width), GUILayout.Height(PillHeight));

			if (Event.current.type == EventType.Repaint)
			{
				var isHovered = GUI.enabled && rect.Contains(Event.current.mousePosition);
				var shown = GUI.enabled ? fill : Color.Lerp(fill, Palette.Off, 0.6f);
				EditorGUI.DrawRect(rect, isHovered ? Color.Lerp(shown, Color.white, 0.15f) : shown);
			}

			_pillLabelStyle.fontStyle = italic ? FontStyle.Italic : FontStyle.Normal;
			_pillLabelStyle.normal.textColor = fill.grayscale > 0.6f ? new Color(0.12f, 0.12f, 0.12f) : Color.white;
			GUI.Label(rect, content, _pillLabelStyle);

			return GUI.Button(rect, GUIContent.none, GUIStyle.none);
		}

		private bool MatchesSearch(LogCategory channel)
		{
			return _search.Length == 0 || channel.Name.IndexOf(_search, StringComparison.OrdinalIgnoreCase) >= 0;
		}

		//Inside its section a channel drops the section's name: "Acme.Inventory.Save" shows as "Inventory.Save".
		private static string GetShortName(string sectionName, LogCategory channel)
		{
			var prefix = sectionName + ".";
			return channel.Name.StartsWith(prefix, StringComparison.Ordinal) ? channel.Name.Substring(prefix.Length) : channel.Name;
		}

		private static string GetFirstSegment(string name)
		{
			var dot = name.IndexOf('.');
			return dot > 0 ? name.Substring(0, dot) : name;
		}

		private Color GetLevelColor(LogLevel level)
		{
			return level switch
			{
				LogLevel.Trace => new Color(0.7f, 0.7f, 0.7f),
				LogLevel.Debug => new Color(0.6f, 0.8f, 1f),
				LogLevel.Info => new Color(0.6f, 1f, 0.6f),
				LogLevel.Warning => new Color(1f, 0.9f, 0.4f),
				LogLevel.Error => new Color(1f, 0.5f, 0.4f),
				LogLevel.Fatal => new Color(1f, 0.3f, 0.3f),
				_ => Color.white,
			};
		}

		private void ToggleEnableLogs()
		{
			var defines = LogEditorUtility.GetCurrentBuildProfileDefines(out var namedBuildTarget);
			var definesList = defines.Split(';').ToList();

			if (_logsEnabled)
			{
				if (!definesList.Contains(LogEditorConstants.LogsEnabledSymbol))
				{
					definesList.Add(LogEditorConstants.LogsEnabledSymbol);
				}
			}
			else
			{
				definesList.Remove(LogEditorConstants.LogsEnabledSymbol);
			}

			var newDefines = string.Join(";", definesList);
			PlayerSettings.SetScriptingDefineSymbols(namedBuildTarget, newDefines);
			Debug.Log($"[QBS] Logging {(_logsEnabled ? "enabled" : "disabled")}");
		}

		private void WriteMinimumLevel(LogLevel level)
		{
			Log.MinimumLevel = level;
			EditorPrefs.SetInt(LogEditorConstants.MinimumLevel, (int)level);

			if (_runtimeSettings != null)
			{
				_runtimeSettings.MinimumLevel = level;
				EditorUtility.SetDirty(_runtimeSettings);
				AssetDatabase.SaveAssets();
			}

			Debug.Log($"[QBS] Minimum log level set to: {level}");
		}

		/// <summary>Saves the live rules to this machine's settings and to the shipped asset.</summary>
		private void WriteRules()
		{
			var rules = LogRules.All;
			LogEditorUtility.WriteRules(rules);

			if (_runtimeSettings != null)
			{
				_runtimeSettings.Rules = rules;
				EditorUtility.SetDirty(_runtimeSettings);
				AssetDatabase.SaveAssets();
			}
		}

		private void SetStateOfAllChannels(bool enabled)
		{
			Log.SetAllChannelsEnabled(enabled);
			WriteRules();
		}

		private void RefreshChannels()
		{
			var report = LogCategoryValidator.Run();
			_problems = report.Problems;
			_editorOnlyChannels = report.EditorOnly;

			var all = LogCategory.All;
			_individualChannels = all.Where(c => !c.IsGroup).OrderBy(c => c.Name, StringComparer.Ordinal).ToList();
			_compositeChannels = all.Where(c => c.IsGroup).OrderBy(c => c.Name, StringComparer.Ordinal).ToList();

			//Core's own channels get their own section; everyone else's are grouped by the first part of the name,
			//which is their owner's namespace.
			var builtIns = new HashSet<LogCategory>
			(
				typeof(LogChannel).GetFields(BindingFlags.Public | BindingFlags.Static)
					.Select(field => field.GetValue(null))
					.OfType<LogCategory>()
			);

			_sections = _individualChannels
				.GroupBy(c => builtIns.Contains(c) ? BuiltInSection : GetFirstSegment(c.Name))
				.OrderBy(g => g.Key == BuiltInSection ? 0 : 1)
				.ThenBy(g => g.Key, StringComparer.Ordinal)
				.Select(g => (g.Key, g.ToList()))
				.ToList();
		}

		private static string GetRuleTooltip(LogCategory channel)
		{
			var rule = LogRules.GetMatchingRule(channel.Name);
			var state = rule.Enabled ? "on" : "off";

			if (rule.Prefix == channel.Name)
			{
				return $"{channel.Name}: {state} by its own rule";
			}

			return rule.Prefix == LogRules.Root
				? $"{channel.Name}: {state}, following the root rule"
				: $"{channel.Name}: {state}, following '{rule.Prefix}'";
		}

		private void SyncRuntimeSettingsFromEditorPrefs()
		{
			if (_runtimeSettings == null)
			{
				return;
			}

			_runtimeSettings.MinimumLevel = _minimumLevel;
			_runtimeSettings.Rules = LogRules.All;
			EditorUtility.SetDirty(_runtimeSettings);
			AssetDatabase.SaveAssets();
		}

		private static class Palette
		{
			public static Color On => EditorGUIUtility.isProSkin ? new Color(0.24f, 0.56f, 0.38f) : new Color(0.36f, 0.72f, 0.48f);
			public static Color Off => EditorGUIUtility.isProSkin ? new Color(0.30f, 0.30f, 0.30f) : new Color(0.80f, 0.80f, 0.80f);
			public static Color Partial => new Color(0.90f, 0.68f, 0.28f);
			public static Color Danger => new Color(0.74f, 0.30f, 0.27f);
			public static Color Card => EditorGUIUtility.isProSkin ? new Color(0.20f, 0.20f, 0.20f) : new Color(0.87f, 0.87f, 0.87f);
		}
	}
}
